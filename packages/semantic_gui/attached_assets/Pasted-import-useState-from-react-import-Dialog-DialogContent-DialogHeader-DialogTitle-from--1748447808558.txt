import { useState } from "react";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Textarea } from "@/components/ui/textarea";
import { Badge } from "@/components/ui/badge";
import { useCreateNode, useGraphData, useCreateEdge } from "@/hooks/use-graph"; // Added useCreateEdge
import { useRetroSounds } from "@/hooks/useRetroSounds";
import { useFMSounds } from "@/hooks/useFMSounds";
import { useToast } from "@/hooks/use-toast";
import { Brain, Sparkles, Zap, Plus, Code, Target } from "lucide-react";

interface AIArchitectProps {
  isOpen: boolean;
  onClose: () => void;
  selectedProjectId: string;
  selectedTemplate: any;
}

interface AnalysisResult {
  request: string;
  existingNodesCount: number;
  suggestedNodes: any[];
  suggestedConnections: any[];
  reasoning: string;
  // Add any other relevant fields from aiResult if needed
}

export function AIArchitect({ isOpen, onClose, selectedProjectId, selectedTemplate }: AIArchitectProps) {
  const [request, setRequest] = useState("");
  const [isProcessing, setIsProcessing] = useState(false);
  const [analysis, setAnalysis] = useState<AnalysisResult | null>(null);
  
  const { data: graphData } = useGraphData(selectedProjectId);
  const createNodeMutation = useCreateNode();
  const createEdgeMutation = useCreateEdge(); // Instantiated useCreateEdge
  const { playSound } = useRetroSounds();
  const { playGlitch } = useFMSounds();
  const { toast } = useToast();

  // AI Analysis Engine
  const analyzeRequest = async () => {
    if (!request.trim()) return;
    
    setIsProcessing(true);
    playSound('success');

    const existingNodes = graphData?.nodes || [];
    const existingEdges = graphData?.edges || [];

    const payload = {
      request: request.trim(),
      existingNodes,
      existingEdges,
      templateId: selectedTemplate?.id || "nestjs-hexagonal",
    };

    try {
      const response = await fetch(`/api/projects/${selectedProjectId}/ai-analyze-request`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(payload),
      });

      if (!response.ok) {
        const errorData = await response.json();
        throw new Error(errorData.error || 'AI analysis failed');
      }

      const aiResult = await response.json();
      
      console.log("🤖 AI Result received:", aiResult);
      
      // Handle case where suggestedNodes might be empty but reasoning exists
      const nodes = aiResult.suggestedNodes || [];
      const connections = aiResult.suggestedConnections || [];
      
      console.log("📊 Parsed nodes:", nodes.length, "connections:", connections.length);
      
      setAnalysis({
        request: request.trim(),
        existingNodesCount: existingNodes.length,
        suggestedNodes: nodes,
        suggestedConnections: connections,
        reasoning: aiResult.reasoning || `AI analysis complete for "${request.trim()}". ${nodes.length === 0 ? 'No new nodes suggested - architecture may be complete.' : ''}`,
      });
      // playSound('nodeCreate'); // Moved to createSuggestedNodes or consider a different sound for analysis success
    } catch (error: any) {
      console.error("AI analysis request failed:", error);
      toast({
        title: "Error",
        description: error.message || "Failed to get AI suggestions.",
        variant: "destructive",
      });
      setAnalysis(null);
    } finally {
      setIsProcessing(false);
    }
  };

  // Create all suggested nodes and connections with epic effects
  const applyAISuggestions = async () => { // Renamed function
    if (!analysis) {
      toast({ title: "Error", description: "No analysis data available.", variant: "destructive" });
      return;
    }
    
    if (!analysis.suggestedNodes?.length && !analysis.suggestedConnections?.length) {
      toast({ 
        title: "🤖 AI Analysis Complete", 
        description: "No new nodes needed - your architecture looks complete for this request!", 
        variant: "default" 
      });
      return;
    }
    
    playSound('success');
    playGlitch();
    setIsProcessing(true);

    const nodeNameOrTempIdToActualId = new Map<string, string>();

    // Populate with existing node names and IDs
    (graphData?.nodes || []).forEach(node => {
      if (node.id && node.name) { // Ensure node.id and node.name are not undefined
          nodeNameOrTempIdToActualId.set(node.name, node.id);
      }
    });

    let createdNodesCount = 0;
    if (analysis.suggestedNodes) {
      for (let i = 0; i < analysis.suggestedNodes.length; i++) {
        const nodeSuggestion = analysis.suggestedNodes[i];
        
        const nodeData = {
          name: nodeSuggestion.name,
          type: nodeSuggestion.type,
          projectId: selectedProjectId,
          templateId: selectedTemplate?.id || "nestjs-hexagonal",
          position: { 
            x: 200 + (i * 150) + Math.random() * 50, // Add some jitter to position
            y: 100 + (i * 100) + Math.random() * 50  // Add some jitter to position
          },
          description: nodeSuggestion.description,
          metadata: {
            suggestedCode: [nodeSuggestion.suggestedCode || ""],
            aiGenerated: [true],
            functionality: [request || "AI Suggested"]
          }
        };

        try {
          await new Promise(resolve => setTimeout(resolve, 300)); // Dramatic pause
          // Assuming createNodeMutation.mutateAsync returns the created node object with its id
          const createdNode = await createNodeMutation.mutateAsync(nodeData);
          
          if (createdNode && createdNode.id && createdNode.name) {
            nodeNameOrTempIdToActualId.set(createdNode.name, createdNode.id); // Use actual name from created node
            createdNodesCount++;
            playSound('nodeCreate');
            toast({
              title: `🎮 AI_NODE_CREATED: ${createdNode.name}`,
              description: `+25 XP • ${nodeSuggestion.description}`,
            });
          } else {
             console.warn("Created node, its ID, or name is undefined", createdNode);
             toast({
                title: "Warning",
                description: `Node ${nodeSuggestion.name} might not have been created correctly or ID/name is missing.`,
                variant: "destructive"
             });
          }
        } catch (error: any) {
          console.error(`Failed to create node ${nodeSuggestion.name}:`, error);
          toast({
            title: "Error Creating Node",
            description: `Could not create node ${nodeSuggestion.name}: ${error.message}`,
            variant: "destructive",
          });
        }
      }
    }
    
    let createdEdgesCount = 0;
    if (analysis.suggestedConnections) {
      for (const conn of analysis.suggestedConnections) {
        const sourceId = nodeNameOrTempIdToActualId.get(conn.from); // 'from' should be node name
        const targetId = nodeNameOrTempIdToActualId.get(conn.to);   // 'to' should be node name

        if (sourceId && targetId) {
          const edgeData = {
            sourceNodeId: sourceId,
            targetNodeId: targetId,
            type: conn.type || "related", // Default type or from suggestion
            label: conn.label || "",
            projectId: selectedProjectId,
            metadata: {
              aiGenerated: true,
              reason: conn.reason || "AI suggested this connection",
            },
          };
          try {
            await new Promise(resolve => setTimeout(resolve, 150)); // Short pause for edge
            await createEdgeMutation.mutateAsync(edgeData);
            createdEdgesCount++;
            // Optional: toast for edge creation
            // playSound('edgeCreate'); // Example: if you have an edge creation sound
          } catch (error: any) {
            console.error('Failed to create edge:', conn, error);
            toast({
              title: "Error Creating Edge",
              description: `Could not connect ${conn.from} to ${conn.to}: ${error.message}`,
              variant: "destructive",
            });
          }
        } else {
          console.warn('Could not map node names to IDs for connection:', conn, 'Mapped sourceId:', sourceId, 'Mapped targetId:', targetId, 'Available map:', Object.fromEntries(nodeNameOrTempIdToActualId));
           toast({
            title: "Warning",
            description: `Could not create connection between ${conn.from} and ${conn.to}. Node IDs not found.`,
            variant: "default" // Use default or a less intrusive variant
          });
        }
      }
    }

    setIsProcessing(false);
    if (createdNodesCount > 0 || createdEdgesCount > 0) {
      playSound('success');
    }
    
    toast({
      title: "🧠 AI_ARCHITECTURE_COMPLETE!",
      description: `Created ${createdNodesCount} nodes and ${createdEdgesCount} connections. Total XP: ${(createdNodesCount * 25)}`,
    });
    
    onClose();
    setAnalysis(null);
    setRequest("");
  };

  return (
    <Dialog open={isOpen} onOpenChange={onClose}>
      <DialogContent className="w-[95vw] max-w-lg mx-auto bg-card border-2 border-accent hacker-glow">
        <DialogHeader className="pb-4 border-b border-accent/30">
          <DialogTitle className="flex items-center space-x-3 text-accent terminal-font font-black hacker-text-glow">
            <Brain className="h-6 w-6" />
            <span>AI_ARCHITECT</span>
            <div className="text-xs text-primary">
              [NEURAL_NETWORK_ACTIVE]
            </div>
          </DialogTitle>
        </DialogHeader>

        <div className="space-y-6 py-4">
          {/* Request Input */}
          <div className="space-y-3">
            <div className="flex items-center space-x-2">
              <Code className="h-4 w-4 text-accent" />
              <span className="text-accent terminal-font font-black text-sm hacker-text-glow">
                FUNCTIONALITY_REQUEST
              </span>
            </div>
            
            <Textarea
              value={request}
              onChange={(e) => setRequest(e.target.value)}
              placeholder="Describe the functionality you need... (e.g., 'user login system', 'payment processing', 'product catalog')"
              className="bg-input border-2 border-accent text-accent placeholder-accent/50 terminal-font hacker-glow min-h-[100px]"
            />
            
            <div className="text-xs text-accent/70 terminal-font">
              {">"} AI will analyze your existing graph and suggest optimal node architecture
            </div>
          </div>

          {/* Analysis Results */}
          {analysis && (
            <div className="space-y-4 p-4 bg-accent/10 border border-accent/30 rounded">
              <div className="flex items-center space-x-2">
                <Sparkles className="h-4 w-4 text-accent" />
                <span className="text-accent terminal-font font-black text-sm">AI_ANALYSIS</span>
              </div>
              
              <div className="text-xs text-accent/90 terminal-font">
                {analysis.reasoning}
              </div>
              
              {analysis.suggestedNodes && analysis.suggestedNodes.length > 0 ? (
                <div className="space-y-2">
                  {analysis.suggestedNodes.map((node: any, index: number) => (
                    <div key={index} className="flex items-center justify-between p-2 bg-accent/5 border border-accent/20 rounded">
                      <div className="flex-1">
                        <div className="text-sm font-black text-accent terminal-font">{node.name}</div>
                        <div className="text-xs text-accent/70 terminal-font">{node.description}</div>
                      </div>
                      <Badge variant="outline" className="text-xs border-accent text-accent terminal-font">
                        {node.type.toUpperCase()}
                      </Badge>
                    </div>
                  ))}
                </div>
              ) : (
                <div className="p-3 bg-primary/10 border border-primary/30 rounded">
                  <div className="text-sm text-primary terminal-font font-black">
                    🎯 ARCHITECTURE_COMPLETE
                  </div>
                  <div className="text-xs text-primary/70 terminal-font mt-1">
                    No new nodes needed. Your current architecture handles this request.
                  </div>
                </div>
              )}
              
              {analysis.suggestedConnections && analysis.suggestedConnections.length > 0 && (
                <div className="mt-3 space-y-1">
                  <div className="text-xs text-accent/80 terminal-font font-black">SUGGESTED_CONNECTIONS:</div>
                  {analysis.suggestedConnections.map((conn: any, index: number) => (
                    <div key={index} className="text-xs text-accent/70 terminal-font">
                      {`🔗 ${conn.sourceName} → ${conn.targetName} (${conn.label || 'related'})`}
                    </div>
                    </div>
                  ))}
                </div>
              )}
              
              <div className="text-xs text-accent/80 terminal-font mt-2">
                {">"} SUGGESTED_NODES: {analysis.suggestedNodes?.length || 0}, SUGGESTED_CONNECTIONS: {analysis.suggestedConnections?.length || 0}
              </div>
              <div className="text-xs text-accent/80 terminal-font">
                {">"} TOTAL_XP_REWARD: {(analysis.suggestedNodes?.length || 0) * 25} XP
              </div>
            </div>
          )}

          {/* Existing Graph Info */}
          {graphData && (
            <div className="p-3 bg-primary/10 border border-primary/30 rounded">
              <div className="text-xs text-primary terminal-font">
                EXISTING_GRAPH: {graphData.nodes?.length || 0} nodes, {graphData.edges?.length || 0} connections
              </div>
            </div>
          )}
        </div>

        {/* Action Buttons */}
        <div className="flex space-x-3 pt-4 border-t border-accent/30">
          {!analysis ? (
            <>
              <Button
                onClick={onClose}
                variant="outline"
                className="flex-1 border-2 border-red-400 text-red-400 hover:bg-red-400/20 terminal-font font-black"
              >
                CANCEL
              </Button>
              <Button
                onClick={analyzeRequest}
                disabled={!request.trim() || isProcessing}
                className="flex-1 bg-accent text-black hover:bg-accent/80 border-2 border-accent terminal-font font-black hacker-glow"
              >
                {isProcessing ? (
                  <div className="animate-spin h-4 w-4 mr-2 border-2 border-black border-t-transparent rounded-full" />
                ) : (
                  <Brain className="h-4 w-4 mr-2" />
                )}
                ANALYZE
              </Button>
            </>
          ) : (
            <>
              <Button
                onClick={() => {
                  setAnalysis(null);
                  setRequest("");
                }}
                variant="outline"
                className="flex-1 border-2 border-accent text-accent hover:bg-accent/20 terminal-font font-black"
              >
                MODIFY
              </Button>
              <Button
                onClick={() => {
                  // FM glitch sound effect
                  const audioContext = new (window.AudioContext || (window as any).webkitAudioContext)();
                  const carrier1 = audioContext.createOscillator();
                  const carrier2 = audioContext.createOscillator();
                  const modulator1 = audioContext.createOscillator();
                  const modulator2 = audioContext.createOscillator();
                  const subOsc = audioContext.createOscillator();
                  
                  const carrierGain1 = audioContext.createGain();
                  const carrierGain2 = audioContext.createGain();
                  const modGain1 = audioContext.createGain();
                  const modGain2 = audioContext.createGain();
                  const subGain = audioContext.createGain();
                  const masterGain = audioContext.createGain();
                  const filter = audioContext.createBiquadFilter();
                  
                  // Glitch FM routing
                  modulator1.connect(modGain1);
                  modulator2.connect(modGain2);
                  modGain1.connect(carrier1.frequency);
                  modGain2.connect(carrier2.frequency);
                  
                  carrier1.connect(carrierGain1);
                  carrier2.connect(carrierGain2);
                  subOsc.connect(subGain);
                  
                  carrierGain1.connect(filter);
                  carrierGain2.connect(filter);
                  subGain.connect(filter);
                  filter.connect(masterGain);
                  masterGain.connect(audioContext.destination);
                  
                  // Rapid pitch sweeps for glitch effect
                  carrier1.type = 'square';
                  carrier2.type = 'sawtooth';
                  subOsc.type = 'triangle';
                  
                  carrier1.frequency.setValueAtTime(440, audioContext.currentTime);
                  carrier1.frequency.exponentialRampToValueAtTime(110, audioContext.currentTime + 0.05);
                  carrier1.frequency.exponentialRampToValueAtTime(880, audioContext.currentTime + 0.1);
                  
                  carrier2.frequency.setValueAtTime(330, audioContext.currentTime);
                  carrier2.frequency.exponentialRampToValueAtTime(165, audioContext.currentTime + 0.1);
                  
                  modGain1.gain.setValueAtTime(200, audioContext.currentTime);
                  modGain2.gain.setValueAtTime(150, audioContext.currentTime);
                  
                  filter.frequency.setValueAtTime(2000, audioContext.currentTime);
                  filter.frequency.exponentialRampToValueAtTime(500, audioContext.currentTime + 0.1);
                  
                  masterGain.gain.setValueAtTime(0.2, audioContext.currentTime);
                  masterGain.gain.exponentialRampToValueAtTime(0.01, audioContext.currentTime + 0.1);
                  
                  [carrier1, carrier2, modulator1, modulator2, subOsc].forEach(osc => {
                    osc.start(audioContext.currentTime);
                    osc.stop(audioContext.currentTime + 0.1);
                  });
                  
                  applyAISuggestions(); // Updated call site
                }}
                disabled={isProcessing}
                className="flex-1 bg-primary text-black hover:bg-primary/80 border-2 border-primary terminal-font font-black hacker-glow"
              >
                {isProcessing ? (
                  <div className="animate-spin h-4 w-4 mr-2 border-2 border-black border-t-transparent rounded-full" />
                ) : (
                  <Zap className="h-4 w-4 mr-2" />
                )}
                CREATE_ALL
              </Button>
            </>
          )}
        </div>
      </DialogContent>
    </Dialog>
  );
}
